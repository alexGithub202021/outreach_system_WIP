package main

import (
	"database/sql"
	"fmt"
	"io"
	"log"
	"os"
	"time"

	"github.com/joho/godotenv"

	_ "github.com/lib/pq"
)

func main() {

	stdLog("-- start go api --\n", 4, "app/logs/postgre_conn.log")

	testPostgreConnection()

}

func stdLog(msg string, level int8, path string) {

	m := make(map[int]string) // creates a map with string keys and integer values
	m[1] = "INFO"
	m[2] = "DEBUG"
	m[3] = "WARN"
	m[4] = "ERROR"

	file, err := os.OpenFile(path, os.O_CREATE|os.O_WRONLY|os.O_APPEND, 0666)
	if err != nil {
		log.Println("ERROR Failed to open log file:", err)
		return
	}
	defer file.Close()

	// Combine stdout and file into a single writer
	multiWriter := io.MultiWriter(os.Stdout, file)

	log.SetOutput(multiWriter)
	log.Println(m[int(level)] + " " + msg)
}

func testPostgreConnection() {

	log_path := "app/logs/postgre_conn.log"

	if !loadEnv(log_path) {
		return
	}

	dbName, connString, res := connectPostgre(log_path)
	if !res {
		return
	}

	db, res := testConnection(connString, log_path, dbName)
	if !res {
		return
	}

	// Optional: Set connection pool settings
	db.SetMaxIdleConns(25)
	db.SetMaxOpenConns(100)
	db.SetConnMaxLifetime(time.Hour)

	defer db.Close()

	stdLog("--- Fetching users ---", 1, log_path)

	rows, err, res := queryDb(db, log_path)
	if !res {
		return
	}

	if !processRows(rows, log_path, err) {
		return
	}

}

func queryDb(db *sql.DB, log_path string) (*sql.Rows, error, bool) {
	selectQuery := `SELECT login, nom, email FROM "user" LIMIT 10`

	rows, err := db.Query(selectQuery)
	if err != nil {
		msg := fmt.Sprintf("Error querying users: %v", err)
		stdLog(msg, 4, log_path)
		return nil, nil, false
	}
	defer rows.Close()
	return rows, err, true
}

func testConnection(connString string, log_path string, dbName string) (*sql.DB, bool) {
	db, err := sql.Open("postgres", connString)
	if err != nil {
		msg := fmt.Sprintf("Error opening connection: %v", err)
		stdLog(msg, 4, log_path)
		return nil, false
	}

	if err := db.Ping(); err != nil {
		msg := fmt.Sprintf("Error pinging database: %v", err)
		stdLog(msg, 4, log_path)
		return nil, false
	}

	stdLog("Successfully connected to database: "+dbName, 1, log_path)
	return db, true
}

func connectPostgre(log_path string) (string, string, bool) {
	dbHost := os.Getenv("DB_HOST")
	dbPort := os.Getenv("DB_PORT")
	dbUser := os.Getenv("DB_USER")
	dbPass := os.Getenv("DB_PASS")
	dbName := os.Getenv("DB_NAME")

	stdLog("DB connection params -> DB_HOST: "+dbHost+" DB_PORT: "+dbPort+" DB_USER: "+dbUser+" DB_PASS: "+dbPass+" DB_NAME: "+dbName, 1, log_path)

	if dbHost == "" || dbPort == "" || dbUser == "" || dbPass == "" || dbName == "" {
		stdLog("Missing required database environment variables. Please set DB_HOST, DB_PORT, DB_USER, DB_PASS, and DB_NAME.", 4, log_path)
		return "", "", false
	}

	connString := fmt.Sprintf("host=%s port=%s user=%s password=%s dbname=%s sslmode=disable",
		dbHost, dbPort, dbUser, dbPass, dbName)
	return dbName, connString, true
}

func processRows(rows *sql.Rows, log_path string, err error) bool {
	for rows.Next() {
		var login string
		var nom string
		var email string

		if err := rows.Scan(&login, &nom, &email); err != nil {
			msg := fmt.Sprintf("Error scanning row: %v", err)
			stdLog(msg, 4, log_path)
			return false
		}
		msg := fmt.Sprintf("Login: %s, Name: %s, Email: %s", login, nom, email)
		stdLog(msg, 1, log_path)

	}

	if err = rows.Err(); err != nil {
		msg := fmt.Sprintf("Row iteration error: %v", err)
		stdLog(msg, 4, log_path)
		return false
	}
	return true
}

func loadEnv(log_path string) bool {
	err := godotenv.Load("app/config/.env")
	if err != nil {
		if err.Error() == "no such file or directory" {
			stdLog("No .env file found. Continuing with defaults.", 4, log_path)
			return false
		} else {
			msg := fmt.Sprintf("Failed to load .env file: %v", err)
			stdLog(msg, 4, log_path)
			return false
		}
	}
	return true
}

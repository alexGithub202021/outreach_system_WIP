func stdLog(msg string, level int8, path string) {

	m := make(map[int]string) // creates a map with string keys and integer values
	m[1] = "INFO"
	m[2] = "DEBUG"
	m[3] = "WARN"
	m[4] = "ERROR"

	file, err := os.OpenFile(path, os.O_CREATE|os.O_WRONLY|os.O_APPEND, 0666)
	if err != nil {
		log.Println("ERROR Failed to open log file:", err)
		return
	}
	defer file.Close()

	// Combine stdout and file into a single writer
	multiWriter := io.MultiWriter(os.Stdout, file)

	log.SetOutput(multiWriter)
	log.Println(m[int(level)] + " " + msg)
}

func readStream(conn *websocket.Conn, logger *log.Logger) error {
	for {
		_, msg, err := conn.ReadMessage()
		if err != nil {
			// If connection is broken, try to reconnect
			if websocket.IsCloseError(err, websocket.CloseGoingAway, websocket.CloseAbnormalClosure) {
				log.Println("Connection closed, attempting to reconnect in 3 seconds...")
				time.Sleep(3 * time.Second)

				var err error
				conn, err = reconnect(conn)
				if err != nil {
					log.Printf("Failed to reconnect: %v. Stopping.\n", err)
					return err
				}
				logger.Println("Reconnected successfully.")
			}
			return err
		}

		// Parse the JSON message
		var data map[string]interface{}
		if err := unmarshalMsg(msg, &data); err != nil {
			continue // Skip malformed messages
		}

		// Extract relevant fields
		p := data["P"].(float64) // Current price
		c := data["c"].(float64) // Closing price
		high := data["h"].(float64)
		low := data["l"].(float64)
		volume := data["v"].(float64)

		logger.Printf("Price: %.8f | Vol: %.2f | High: %.8f | Low: %.8f | Close: %.8f", p, volume, high, low, c)
	}
}

func reconnect(conn *websocket.Conn) (*websocket.Conn, error) {
	// We must recreate the dial connection.
	// Note: In a production app, you should add heartbeat logic here.
	newConn, _, err := websocket.DefaultDialer.Dial(wsURL, nil)
	if err != nil {
		return nil, err
	}
	return newConn, nil
}

func unmarshalMsg(msg []byte, data interface{}) error {
	return json.Unmarshal(msg, data)
}

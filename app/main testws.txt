var operation = "aggTrade"
var url = fmt.Sprintf("wss://stream.binance.com:9443/ws/btcusdt@%s", operation)
var logPath = "app/logs/ws_conn_log.json"

func main() {

	cleanup, err := logger.InitGlobalLogger(logPath)
	if nil != err {
		panic("Failed to initialize logger: " + err.Error())
	}
	defer cleanup()

	logger.Log.Info("go app is running")

	testSleep(1800)

	// testWebSockets()

}

func readStream(conn *websocket.Conn) error {

	for {

		_, msg, err := conn.ReadMessage()

		if err != nil {
			// cases when the server closes the connection "gracefully"
			if _, ok := err.(*websocket.CloseError); ok {
				logger.Log.Error("Server closed connection. Exiting.")
				return nil
			}

			logger.Log.Error("Error details", slog.String("error message", err.Error()))

			// other cases
			conn, err = reconnect()
			if err != nil {
				logger.Log.Error("Failed to reconnect. Stopping.")
				return err
			}
		}

		// Parse the JSON message
		var data map[string]interface{}
		if err := unmarshalMsg(msg, &data); nil != err {
			continue // Skip malformed messages
		}

		// Extract relevant fields
		s := data["s"].(string) // symbol
		p := data["p"].(string) // Current price
		logger.Log.Info("asset price", slog.String(s, p))

	}
}

func reconnect() (*websocket.Conn, error) {
	// We must recreate the dial connection.
	// Note: In a production app, you should add heartbeat logic here.
	newConn, _, err := websocket.DefaultDialer.Dial(url, nil)
	if nil != err {
		return nil, err
	}
	return newConn, nil
}

func unmarshalMsg(msg []byte, data interface{}) error {
	return json.Unmarshal(msg, data)
}

func testWebSockets() {

	dialer := tlsConfig()

	// establish ws connection
	conn, _, err := dialer.Dial(url, http.Header{})
	if nil != err {
		logger.Log.Error("Dial error", slog.String("error message", err.Error()))
	}
	defer conn.Close()

	logger.Log.Info("Connected to Binance WebSocket")

	if nil != readStream(conn) {
		logger.Log.Error("Stream ended", slog.String("error message", err.Error()))
	}
}

func tlsConfig() websocket.Dialer {
	certPool, err := x509.SystemCertPool()
	if err != nil {
		logger.Log.Error("Failed to read CA bundle", slog.String("error message", err.Error()))
	}

	tlsConfig := &tls.Config{ // Assign the address
		RootCAs:    certPool,
		ServerName: "stream.binance.com",
	}

	dialer := websocket.Dialer{
		// Create a custom dialer with insecure TLS (DEV ONLY!)
		// TLSClientConfig: &tls.Config{InsecureSkipVerify: true},
		TLSClientConfig: tlsConfig,
		Proxy:           http.ProxyFromEnvironment,
	}
	return dialer
}

func getLogger(logPath string) (*slog.Logger, func(), error) {
	file, err := os.OpenFile(logPath, os.O_CREATE|os.O_WRONLY|os.O_APPEND, 0666)
	if err != nil {
		return nil, nil, err
	}

	// Pass the file to the JSON handler
	logger := slog.New(slog.NewJSONHandler(file, nil))

	cleanup := func() {
		file.Close()
	}

	return logger, cleanup, nil
}

func testSleep(dur int) {

	duration := time.Duration(dur)

	fmt.Println("Starting...")
	time.Sleep(duration * time.Second) // Pause for 5 seconds
	fmt.Println("Resuming...")
}
